1. Tạo cơ sở dữ liệu và các bảng

Bài làm sử dụng cú pháp SQL Server. Nếu dùng MySQL thì có thể bỏ chữ GO và đổi NVARCHAR thành VARCHAR hoặc TEXT tùy yêu cầu. 
CREATE DATABASE QuanLyNhanSu;
GO

USE QuanLyNhanSu;
GO

CREATE TABLE PHONGBAN (
    PHG INT PRIMARY KEY,
    TENPHG NVARCHAR(50)
);

CREATE TABLE NHANVIEN (
    HONV NVARCHAR(30),
    TENLOT NVARCHAR(30),
    TENNV NVARCHAR(30),
    PHAI NVARCHAR(5),
    LUONG FLOAT,
    MANV CHAR(9) PRIMARY KEY,
    NGSINH DATE,
    DIACHI NVARCHAR(100),
    PHG INT,
    FOREIGN KEY (PHG) REFERENCES PHONGBAN(PHG)
);

CREATE TABLE DEAN (
    TENDA NVARCHAR(50),
    MADA INT PRIMARY KEY,
    DDIEM_DA NVARCHAR(50),
    PHG INT,
    FOREIGN KEY (PHG) REFERENCES PHONGBAN(PHG)
);

CREATE TABLE PHANCONG (
    MANV CHAR(9),
    MADA INT,
    SOGIO FLOAT,
    PRIMARY KEY (MANV, MADA),
    FOREIGN KEY (MANV) REFERENCES NHANVIEN(MANV),
    FOREIGN KEY (MADA) REFERENCES DEAN(MADA)
);

Câu 2Hiển thị tất cả thông tin của bảng NHANVIEN.
SELECT *
FROM NHANVIEN;
Câu 3Hiển thị thông tin của những nhân viên ở phòng số 5, sắp thứ tự theo tên nhân viên giảm dần.
SELECT *
FROM NHANVIEN
WHERE PHG = 5
ORDER BY TENNV DESC;
Câu 4Hiển thị mã nhân viên, họ nhân viên, tên lót và tên nhân viên của những nhân viên ở phòng số 5 và có lương từ 3000 trở lên.
SELECT MANV, HONV, TENLOT, TENNV
FROM NHANVIEN
WHERE PHG = 5 AND LUONG >= 3000;
Câu 5Hiển thị mã nhân viên, tên nhân viên của những nhân viên có lương từ 3000 đến 5000.
SELECT MANV, TENNV
FROM NHANVIEN
WHERE LUONG BETWEEN 3000 AND 5000;

4. Trả lời câu 6 đến câu 10

Câu 6Hiển thị thông tin của những nhân viên ở địa chỉ có tên đường là Nguyễn.
SELECT *
FROM NHANVIEN
WHERE DIACHI LIKE N'Nguyễn%';
Câu 7Cho biết số lượng nhân viên.
SELECT COUNT(*) AS SoLuongNhanVien
FROM NHANVIEN;
Câu 8Cho biết số lượng nhân viên trong từng phòng ban.
SELECT PB.PHG, PB.TENPHG, COUNT(NV.MANV) AS SoLuongNhanVien
FROM PHONGBAN PB
LEFT JOIN NHANVIEN NV ON PB.PHG = NV.PHG
GROUP BY PB.PHG, PB.TENPHG;
Câu 9Cho biết số lượng nhân viên trong từng phòng ban, chỉ hiển thị những phòng ban có từ 2 nhân viên trở lên.
SELECT PB.PHG, PB.TENPHG, COUNT(NV.MANV) AS SoLuongNhanVien
FROM PHONGBAN PB
JOIN NHANVIEN NV ON PB.PHG = NV.PHG
GROUP BY PB.PHG, PB.TENPHG
HAVING COUNT(NV.MANV) >= 2;
Câu 10Hiển thị thông tin về mã nhân viên, tên nhân viên và tên phòng ban ở phòng kế toán.
SELECT NV.MANV, NV.TENNV, PB.TENPHG
FROM NHANVIEN NV
JOIN PHONGBAN PB ON NV.PHG = PB.PHG
WHERE PB.TENPHG = N'Kế toán';

Câu 12Sửa thông tin của nhân viên mới Lê Ngọc Yến là thuộc Phòng 4.
UPDATE NHANVIEN
SET PHG = 4
WHERE MANV = '987654321';
Câu 13Xóa các phòng ban không có nhân viên.
DELETE FROM PHONGBAN
WHERE PHG NOT IN (
    SELECT DISTINCT PHG
    FROM NHANVIEN
    WHERE PHG IS NOT NULL
);
Câu 14Xóa các đề án chưa phân công nhân viên.
DELETE FROM DEAN
WHERE MADA NOT IN (
    SELECT DISTINCT MADA
    FROM PHANCONG
);
Câu 15Hiển thị thông tin các nhân viên phòng Kế toán, bao gồm: Mã nhân viên, Họ và tên, Phái, Lương và Tên phòng.
SELECT
    NV.MANV,
    NV.HONV + N' ' + NV.TENLOT + N' ' + NV.TENNV AS HoVaTen,
    NV.PHAI,
    NV.LUONG,
    PB.TENPHG
FROM NHANVIEN NV
JOIN PHONGBAN PB ON NV.PHG = PB.PHG
WHERE PB.TENPHG = N'Kế toán';
